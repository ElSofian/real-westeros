on_title_gain = {
    on_actions = { rw_g3_trace_gain }
}
rw_g3_trace_gain = {
    trigger = { rw_g3_trace_title_callback_trigger = yes }
    effect = {
        if = {
            limit = { rw_g3_trace_title_callback_trigger = yes }
            scope:title = {
                debug_log = "RW_G3 callback=gain holder=[ROOT.Char.GetID] title=[THIS.Title.GetNameNoTooltip]"
            }
        }
    }
