# Base mod fails closed. A locally generated PRIVATE adapter supplies native guards.
rw_nd_adapter_available_trigger = { always = no }
rw_nd_native_eligible_trigger = { always = no }

# Original conservative domain: dignity only; no third party's holdings or vassals.
rw_nd_domain_trigger = {
    rw_rp_pending_trigger = yes
    this = scope:rw_rs_holder
    rw_rp_parties_trigger = yes
    is_independent_ruler = yes
    government_has_flag = government_is_feudal
    highest_held_title_tier >= tier_kingdom
    is_at_war = no
    has_active_diarchy = no
    NOT = { has_partition_succession_realm_law_trigger = yes }
    exists = house
    scope:rw_rs_requester = {
        liege = scope:rw_rs_holder
        highest_held_title_tier = tier_county
        government_has_flag = government_is_feudal
        is_at_war = no
        has_active_diarchy = no
        faith = scope:rw_rs_holder.faith
        culture = scope:rw_rs_holder.culture
        exists = house
        NOT = { this = scope:rw_rs_holder.player_heir }
    }
    scope:rw_rs_title = {
        holder = scope:rw_rs_holder
        tier = tier_duchy
        is_landless_type_title = no
        is_noble_family_title = no
        is_head_of_faith = no
        is_leased_out = no
        NOT = { this = scope:rw_rs_holder.primary_title }
        NOT = { has_variable = agot_title_blocked_to_give_away }
        NOT = { has_variable = wagered_county }
    }
}
rw_nd_can_deliver_trigger = {
    rw_nd_adapter_available_trigger = yes
    rw_nd_domain_trigger = yes
    scope:rw_rs_record = { NOT = { has_variable = rw_nd_attempt } }
}
# The old panel cannot reopen a consumed attempt or a newer promise revision.
rw_nd_queued_trigger = {
    rw_nd_adapter_available_trigger = yes
    rw_nd_domain_trigger = yes
    scope:rw_rs_record = {
        var:rw_nd_attempt = scope:rw_rp_revision
        var:rw_nd_result = flag:pending
        has_variable = rw_nd_ticket
    }
}
