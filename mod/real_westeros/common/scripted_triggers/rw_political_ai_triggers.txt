# Read-only policy inputs. The character evaluating a choice is always `this`.
# Existing party aliases name original, known parties in both service and case UI.
rw_ai_known_pair_trigger = {
    exists = scope:rw_service_provider
    exists = scope:rw_service_client
    NOT = { scope:rw_service_provider = scope:rw_service_client }
    OR = { this = scope:rw_service_provider this = scope:rw_service_client }
}
rw_ai_trusted_counterpart_trigger = {
    rw_ai_known_pair_trigger = yes
    OR = {
        AND = {
            this = scope:rw_service_client
            opinion = { target = scope:rw_service_provider value >= 25 }
        }
        AND = {
            this = scope:rw_service_provider
            opinion = { target = scope:rw_service_client value >= 25 }
        }
    }
}
rw_ai_hostile_counterpart_trigger = {
    rw_ai_known_pair_trigger = yes
    OR = {
        AND = {
            this = scope:rw_service_client
            opinion = { target = scope:rw_service_provider value <= -25 }
        }
        AND = {
            this = scope:rw_service_provider
            opinion = { target = scope:rw_service_client value <= -25 }
        }
    }
}
# Own outstanding duties are known to the decider; no counterpart's private case read.
rw_ai_existing_commitment_trigger = {
    OR = {
        has_character_flag = rw_service_obligation
        has_character_flag = rw_fv_obligation
    }
}

# Guard the complete saved decision frame before calling family validation.
rw_ai_gold_promise_context_trigger = {
    trigger_if = {
        limit = {
            exists = scope:rw_service_provider
            exists = scope:rw_service_client
            exists = scope:rw_service_serial
            exists = scope:rw_service_revision
            exists = scope:rw_service_price
            exists = scope:rw_service_mode
            exists = scope:rw_service_decision_serial
        }
        this = scope:rw_service_provider
    }
    trigger_else = { always = no }
}
rw_ai_favor_promise_context_trigger = {
    trigger_if = {
        limit = {
            exists = scope:rw_fv_provider
            exists = scope:rw_fv_client
            exists = scope:rw_fv_serial
            exists = scope:rw_fv_revision
            exists = scope:rw_fv_term
            exists = scope:rw_fv_mode
        }
        this = scope:rw_fv_provider
    }
    trigger_else = { always = no }
}

# Missing legacy timing metadata is neutral. Never infer timing from a late view.
rw_ai_gold_promise_urgent_trigger = {
    trigger_if = {
        limit = { rw_ai_gold_promise_context_trigger = yes }
        rw_service_obligation_actionable_trigger = yes
        scope:rw_service_provider = {
            trigger_if = {
                limit = { has_variable = rw_service_ai_timing_serial }
                var:rw_service_ai_timing_serial = scope:rw_service_serial
                NOT = { has_character_flag = rw_service_ai_wait_room }
            }
            trigger_else = { always = no }
        }
    }
    trigger_else = { always = no }
}
rw_ai_favor_promise_urgent_trigger = {
    trigger_if = {
        limit = { rw_ai_favor_promise_context_trigger = yes }
        rw_fd_due_trigger = yes
        scope:rw_fv_provider = {
            trigger_if = {
                limit = { has_variable = rw_fd_ai_timing_serial }
                var:rw_fd_ai_timing_serial = scope:rw_fv_serial
                NOT = { has_character_flag = rw_fd_ai_later_prompt }
            }
            trigger_else = { always = no }
        }
    }
    trigger_else = { always = no }
}
