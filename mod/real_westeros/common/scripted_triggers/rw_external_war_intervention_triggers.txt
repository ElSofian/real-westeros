# Fail closed with an older private adapter or incorrect load order.
rw_xw_native_adapter_available_trigger = { always = no }
# Character scope. Only the original declaring liege has this native capacity.
rw_xw_native_context_trigger = {
    rw_xw_identity_trigger = yes
    this = scope:rw_xw_ruler
    scope:rw_xw_record = { var:rw_xw_role = flag:declaring_liege }
    NOT = { is_participant_in_war = scope:rw_xw_record.var:rw_xw_war }
    scope:rw_xw_attacker = {
        any_character_war = { this = scope:rw_xw_record.var:rw_xw_war }
        NOT = { any_character_war = { count >= 2 } }
    }
    NOT = { has_variable = rw_uw_native_send_record }
    NOT = { has_variable = rw_uw_native_send_serial }
    NOT = { has_variable = rw_xw_native_send_record }
    NOT = { has_variable = rw_xw_native_send_serial }
}
rw_xw_can_queue_native_trigger = {
    rw_xw_native_adapter_available_trigger = yes
    rw_xw_turn_trigger = yes
    rw_xw_native_context_trigger = yes
}
# Immutable queued identity. Current native validity is checked separately.
rw_xw_native_ticket_trigger = {
    exists = scope:rw_xw_record
    exists = scope:rw_xw_serial
    exists = scope:rw_xw_ruler
    exists = scope:rw_xw_attacker
    exists = scope:rw_xw_defender
    exists = scope:rw_xn_war
    exists = scope:rw_xn_authority
    this = scope:rw_xw_ruler
    scope:rw_xw_record = {
        story_type = story_rw_external_war_response
        story_owner = scope:rw_xw_attacker
        var:rw_xw_schema = 1
        var:rw_xw_serial = scope:rw_xw_serial
        var:rw_xw_ruler = scope:rw_xw_ruler
        var:rw_xw_attacker = scope:rw_xw_attacker
        var:rw_xw_defender = scope:rw_xw_defender
        var:rw_xw_war = scope:rw_xn_war
        var:rw_xw_authority = scope:rw_xn_authority
        var:rw_xw_role = flag:declaring_liege
        var:rw_xw_state = flag:native_queued
        exists = var:rw_xw_war
        has_variable = rw_xw_native_queued_on
        has_variable = rw_xw_native_window
        NOT = { has_variable = rw_xw_native_dispatched_on }
    }
}
# Used only in the isolated next-day event, after native inputs are bound.
rw_xw_can_send_native_trigger = {
    rw_xw_native_adapter_available_trigger = yes
    rw_xw_native_ticket_trigger = yes
    rw_xw_native_context_trigger = yes
    scope:actor = scope:rw_xw_ruler
    scope:recipient = scope:rw_xw_attacker
    scope:target = scope:rw_xw_record.var:rw_xw_war
    is_character_interaction_valid = {
        recipient = scope:rw_xw_attacker
        interaction = stop_attacker_vassal_war_interaction
    }
    scope:target = scope:rw_xw_record.var:rw_xw_war
}
# Native callback context. A sender-local transport marker grants no permission.
rw_xw_native_binding_context_trigger = {
    rw_xw_native_adapter_available_trigger = yes
    exists = scope:actor
    exists = scope:recipient
    scope:actor = {
        NOT = { has_variable = rw_uw_native_send_record }
        NOT = { has_variable = rw_uw_native_send_serial }
        has_variable = rw_xw_native_send_record
        has_variable = rw_xw_native_send_serial
        var:rw_xw_native_send_record = {
            rw_xw_context_trigger = yes
            var:rw_xw_ruler = scope:actor
            var:rw_xw_attacker = scope:recipient
            var:rw_xw_serial = scope:actor.var:rw_xw_native_send_serial
            var:rw_xw_role = flag:declaring_liege
            var:rw_xw_state = flag:closed
            var:rw_xw_result = flag:native_requested
            var:rw_xw_native_dispatched_on = current_date
            NOT = { has_variable = rw_xw_native_bound_on }
        }
        NOT = { is_participant_in_war = var:rw_xw_native_send_record.var:rw_xw_war }
    }
    scope:recipient = { NOT = { any_character_war = { count >= 2 } } }
}
