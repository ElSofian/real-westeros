# Original runtime guards; native entry requires the isolated PRIVATE adapter.
# The unmodified native helper remains authoritative for law/contract exceptions.
rw_uw_native_target_adapter_available_trigger = { always = no }
rw_uw_applicable_ban_trigger = {
    exists = scope:actor
    exists = scope:recipient
    NOT = { scope:actor = scope:recipient }
    scope:actor = {
        is_alive = yes is_adult = yes is_landed = yes is_imprisoned = no
        government_has_flag = government_is_feudal
        exists = liege
        liege = { is_alive = yes }
        OR = {
            liege = { has_realm_law_flag = vassal_all_wars_banned }
            AND = {
                exists = scope:recipient.liege
                liege = scope:recipient.liege
                liege = { has_realm_law_flag = vassal_internal_wars_banned }
            }
        }
    }
    scope:recipient = {
        is_alive = yes is_adult = yes is_landed = yes is_imprisoned = no
        government_has_flag = government_is_feudal
    }
    war_declarer_needs_hook_on_liege = yes
}
# The existing political-response path still requires a shared direct liege.
rw_uw_peer_ban_trigger = {
    rw_uw_applicable_ban_trigger = yes
    exists = scope:recipient.liege
    scope:actor = { liege = scope:recipient.liege }
}
# A native window-local send flag, not a character variable or reusable ticket.
rw_uw_selected_defiance_trigger = {
    rw_uw_applicable_ban_trigger = yes
    always = scope:rw_uw_defy
    NOT = { always = scope:hook }
}

# Decision preference ONLY, never permission: native AI must still discover/send
# the authentic interaction. No random roll in repeatedly evaluated UI triggers.
# A player starts unchecked; this does not change either actor's valid choices.
rw_uw_ai_default_defiance_trigger = {
    rw_uw_applicable_ban_trigger = yes
    NOT = { always = scope:hook }
    scope:actor = {
        is_ai = yes
        rw_ai_wd_motivation_value >= rw_ai_wd_caution_value
    }
}

# STORY scope. Dedup survival is independent of CB, roles and present authority.
# Only the original owner's local war list is visited; never follow a replacement.
rw_uw_exact_war_present_trigger = {
    story_type = story_rw_unlawful_war_response
    var:rw_uw_schema = 1
    exists = var:rw_uw_war
    exists = var:rw_uw_attacker
    story_owner = var:rw_uw_attacker
    save_temporary_scope_as = rw_wc_check
    story_owner = {
        any_character_war = { this = scope:rw_wc_check.var:rw_uw_war }
    }
}
# STORY scope. Maintenance closes lost authority/context; never transfers consent.
rw_uw_record_context_trigger = {
    rw_uw_active_record_trigger = yes
    exists = var:rw_uw_issuer
    exists = var:rw_uw_authority
    save_temporary_scope_as = rw_wc_check
    var:rw_uw_issuer = { is_alive = yes is_adult = yes is_landed = yes is_imprisoned = no }
    var:rw_uw_authority = { holder = scope:rw_wc_check.var:rw_uw_issuer }
    var:rw_uw_attacker = {
        is_alive = yes is_adult = yes is_imprisoned = no
        liege = scope:rw_wc_check.var:rw_uw_issuer
    }
    var:rw_uw_defender = { is_alive = yes liege = scope:rw_wc_check.var:rw_uw_issuer }
}

# STORY scope. Only the exact currently active native claim war qualifies.
rw_uw_active_record_trigger = {
    story_type = story_rw_unlawful_war_response
    var:rw_uw_schema = 1
    exists = var:rw_uw_war
    exists = var:rw_uw_attacker
    exists = var:rw_uw_defender
    story_owner = var:rw_uw_attacker
    save_temporary_scope_as = rw_uw_check
    var:rw_uw_attacker = {
        any_character_war = {
            this = scope:rw_uw_check.var:rw_uw_war
            using_cb = claim_cb
            primary_attacker = scope:rw_uw_check.var:rw_uw_attacker
            primary_defender = scope:rw_uw_check.var:rw_uw_defender
        }
    }
}

    scope:rw_uw_attacker = {

        NOT = { any_character_war = { count >= 2 } }
    }
}

# Used ONLY inside the separate next-day event after binding native input scopes.
rw_uw_can_send_native_trigger = {
    rw_uw_native_target_adapter_available_trigger = yes
    rw_uw_native_ticket_trigger = yes
    rw_uw_native_context_trigger = yes
    NOT = { has_variable = rw_xw_native_send_record }
    NOT = { has_variable = rw_xw_native_send_serial }
    scope:actor = scope:rw_uw_issuer
    scope:recipient = scope:rw_uw_attacker
    scope:target = scope:rw_uw_war
    is_character_interaction_valid = {
        recipient = scope:rw_uw_attacker
        interaction = stop_attacker_vassal_war_interaction
    }
    # Do not proceed if native validation replaced the candidate target context.
    scope:target = scope:rw_uw_war
}

# Native interaction callback context. A one-day sender-local transport marker is
# not permission: exact consumed receipt, current roles and one-war guards apply.
rw_uw_native_binding_context_trigger = {
    exists = scope:actor
    exists = scope:recipient
    scope:actor = {
        NOT = { has_variable = rw_xw_native_send_record }
        NOT = { has_variable = rw_xw_native_send_serial }
        has_variable = rw_uw_native_send_record
        has_variable = rw_uw_native_send_serial
        var:rw_uw_native_send_record = {
            rw_uw_record_context_trigger = yes
            var:rw_uw_issuer = scope:actor
            var:rw_uw_attacker = scope:recipient
            var:rw_uw_serial = scope:actor.var:rw_uw_native_send_serial
            var:rw_uw_state = flag:closed
            var:rw_uw_result = flag:native_requested
            var:rw_uw_native_dispatched_on = current_date
        }
    }
    scope:recipient = { NOT = { any_character_war = { count >= 2 } } }
    NOT = { scope:actor = { is_participant_in_war = scope:actor.var:rw_uw_native_send_record.var:rw_uw_war } }
}

# One request per exact personal receipt; re-opening never renews its total life.
rw_uw_can_demand_trigger = {
    rw_uw_issuer_turn_trigger = yes
    scope:rw_uw_record = { NOT = { has_variable = rw_uw_demanded_on } NOT = { has_variable = rw_uw_reviewed_on } }
}
rw_uw_can_reconsider_trigger = {
    rw_uw_response_identity_trigger = yes
    OR = { this = scope:rw_uw_issuer this = scope:rw_uw_attacker }
    scope:rw_uw_record = {
        has_variable = rw_uw_live
        NOT = { has_variable = rw_uw_reviewed_on }
        var:rw_uw_state = flag:closed
        OR = { var:rw_uw_result = flag:acknowledged var:rw_uw_result = flag:refused var:rw_uw_result = flag:unanswered }
    }
}
