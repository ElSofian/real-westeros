# Story scope: one explicit financial counteroffer to the current petition.
# Political rehabilitation only; no native criminal immunity, property or title right.
rw_house_terms_valid_trigger = {
    rw_house_record_active_trigger = yes
    var:rw_house_last_response = flag:terms_offered
    has_variable = rw_house_terms_window
    var:rw_house_terms_price = 50
    exists = var:rw_house_petitioner
    exists = var:rw_house_requested_ruler
    var:rw_house_petitioner = {
        is_alive = yes
        is_adult = yes
        is_imprisoned = no
        exists = house
        house = prev.var:rw_house_target
        house.house_head = this
        NOT = { is_at_war_with = prev.var:rw_house_requested_ruler }
        NOT = { this = prev.var:rw_house_requested_ruler }
    }
    var:rw_house_authority = { exists = holder holder = prev.var:rw_house_requested_ruler }
    var:rw_house_requested_ruler = { rw_house_sovereign_trigger = yes }
}
# Character root: exact current head, reader, record identity and latest revision.
rw_house_terms_turn_trigger = {
    rw_house_current_head_trigger = yes
    scope:rw_house_record = {
        rw_house_terms_valid_trigger = yes
        var:rw_house_petitioner = root
    }
}
rw_house_terms_affordable_trigger = {
    rw_house_terms_turn_trigger = yes
    gold >= 50
}
