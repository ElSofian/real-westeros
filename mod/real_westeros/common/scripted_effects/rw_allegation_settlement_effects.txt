# Story scope, read-only; reply panels keep their captured terms until acted upon.
rw_as_snapshot_effect = {
    rw_contract_snapshot_effect = { PREFIX = as }
}
# Explicit fresh navigation only; never run automatically in a delayed reply event.
rw_as_prepare_effect = {
    if = {
        limit = { rw_as_base_trigger = yes }
        clear_saved_scope = rw_as_revision
        clear_saved_scope = rw_as_price
        if = {
            limit = { scope:rw_case_record = { var:rw_as_schema = 1 has_variable = rw_as_revision has_variable = rw_as_price } }
            scope:rw_case_record = { rw_as_snapshot_effect = yes }
        }
        trigger_event = { id = rw_allegation_settlement.0001 }
    }
}
# Claimant offers withdrawal for fifty gold from the original respondent.
rw_as_propose_effect = {
    if = {
        limit = { rw_as_can_propose_trigger = yes }
        scope:rw_case_record = {
            set_variable = { name = rw_as_revision value = 0 }
            set_variable = { name = rw_as_turn value = flag:provider }
            rw_contract_offer_effect = { PREFIX = as PRICE = 50 }
            rw_as_snapshot_effect = yes
        }
        rw_as_send_reply_effect = yes
    }
}
# Caller is the current proposer/counterer; no mediator learns these private terms.
rw_as_send_reply_effect = {
    if = {
        limit = { this = scope:rw_service_client }
        scope:rw_service_provider = { trigger_event = { id = rw_allegation_settlement.0001 days = 1 } }
    }
    else = { scope:rw_service_client = { trigger_event = { id = rw_allegation_settlement.0001 days = 1 } } }
}
rw_as_counter_effect = {
    if = {
        limit = { rw_as_responder_trigger = yes }
        scope:rw_case_record = {
            rw_contract_counter_effect = { PREFIX = as TURN_A = provider TURN_B = client }
            rw_as_snapshot_effect = yes
        }
        # Neither seven-day offer window nor original case deadline is renewed.
        rw_as_send_reply_effect = yes
    }
}
rw_as_settle_effect = {
    if = {
        limit = { rw_as_can_settle_trigger = yes }
        scope:rw_case_record = {
            rw_contract_consume_effect = { PREFIX = as }
            # Consume both offer and active case BEFORE native effects/notices.
            # Same retention/result as a voluntary withdrawal; statements stay intact.
            remove_variable = rw_case_pending
            set_variable = { name = rw_case_retained value = flag:yes days = 1825 }
            set_variable = { name = rw_case_stage value = flag:closed }
            set_variable = { name = rw_case_result value = flag:withdrawn }
        }
        scope:rw_service_provider = { remove_short_term_gold = scope:rw_as_price }
        scope:rw_service_client = { add_gold = scope:rw_as_price }
        rw_as_notify_effect = yes
    }
}
rw_as_close_effect = {
    if = {
        limit = { rw_as_can_close_trigger = yes }
        scope:rw_case_record = {
            rw_contract_close_effect = { PREFIX = as }
        }
        rw_as_notify_effect = yes
    }
}
rw_as_notify_effect = {
    scope:rw_service_provider = { if = { limit = { is_alive = yes } trigger_event = { id = rw_allegation_settlement.0001 } } }
    scope:rw_service_client = { if = { limit = { is_alive = yes } trigger_event = { id = rw_allegation_settlement.0001 } } }
}
