# STORY local resolution: preserve completed actions, never undo/repay them.
rw_ba_resolve_effect = {
    if = {
        limit = { var:rw_ba_agreement = flag:active }
        if = {
            limit = { OR = { NOT = { var:rw_ba_payer = { is_alive = yes } } NOT = { var:rw_ba_payee = { is_alive = yes } } } }
            if = { limit = { var:rw_ba_gold = flag:pending } set_variable = { name = rw_ba_gold value = flag:unavailable } }
            if = { limit = { var:rw_ba_support = flag:pending } set_variable = { name = rw_ba_support value = flag:unavailable } }
            if = { limit = { var:rw_ba_release = flag:pending } set_variable = { name = rw_ba_release value = flag:unavailable } }
        }
        if = {
            limit = { NOT = { has_variable = rw_ba_due } }
            if = { limit = { var:rw_ba_gold = flag:pending } set_variable = { name = rw_ba_gold value = flag:expired } }
            if = { limit = { var:rw_ba_support = flag:pending } set_variable = { name = rw_ba_support value = flag:expired } }
            if = { limit = { var:rw_ba_release = flag:pending } set_variable = { name = rw_ba_release value = flag:expired } }
        }
        if = {
            limit = { var:rw_ba_gold = flag:done var:rw_ba_support = flag:done var:rw_ba_release = flag:done }
            set_variable = { name = rw_ba_agreement value = flag:fulfilled }
        }
        else_if = {
            limit = { NOR = { var:rw_ba_gold = flag:pending var:rw_ba_support = flag:pending var:rw_ba_release = flag:pending } }
            set_variable = { name = rw_ba_agreement value = flag:partial }
        }
        if = {
            limit = { NOT = { var:rw_ba_agreement = flag:active } }
            remove_variable = rw_ba_due
            set_variable = { name = rw_ba_finished_on value = current_date }
        }
    }
}
rw_ba_retire_effect = {
    if = {
        limit = { story_type = story_rw_multi_action_agreement }
        save_scope_as = rw_ba_record
        if = { limit = { has_global_variable_list = rw_multi_action_agreements } remove_list_global_variable = { name = rw_multi_action_agreements target = scope:rw_ba_record } }
        end_story = yes
    }
}
# Only either original party may hold the record; storage conveys no action rights.
rw_ba_rehome_effect = {
    clear_saved_scope = rw_ba_custodian
    if = {
        limit = { var:rw_ba_payer = { is_alive = yes NOT = { this = scope:rw_ba_departed } } }
        var:rw_ba_payer = { save_scope_as = rw_ba_custodian }
    }
    else_if = {
        limit = { var:rw_ba_payee = { is_alive = yes NOT = { this = scope:rw_ba_departed } } }
        var:rw_ba_payee = { save_scope_as = rw_ba_custodian }
    }
    if = { limit = { exists = scope:rw_ba_custodian NOT = { story_owner = scope:rw_ba_custodian } } make_story_owner = scope:rw_ba_custodian }
}

}
rw_ba_owner_died_effect = {
    if = {
        limit = { rw_ba_record_trigger = yes }
        story_owner = { save_scope_as = rw_ba_departed }
        # Native alive flags can lag the death callback. Explicit known death
        # invalidates pending acts but preserves completed/attempted/refused receipts.
        if = {
            limit = {
                var:rw_ba_agreement = flag:active
                OR = { var:rw_ba_payer = scope:rw_ba_departed var:rw_ba_payee = scope:rw_ba_departed }
            }
            if = { limit = { var:rw_ba_gold = flag:pending } set_variable = { name = rw_ba_gold value = flag:unavailable } }
            if = { limit = { var:rw_ba_support = flag:pending } set_variable = { name = rw_ba_support value = flag:unavailable } }
            if = { limit = { var:rw_ba_release = flag:pending } set_variable = { name = rw_ba_release value = flag:unavailable } }
            rw_ba_resolve_effect = yes
        }
        if = {
            limit = {
                var:rw_ba_agreement = flag:negotiating
                OR = { var:rw_ba_payer = scope:rw_ba_departed var:rw_ba_payee = scope:rw_ba_departed }
            }
            rw_contract_close_effect = { PREFIX = ba }
            set_variable = { name = rw_ba_agreement value = flag:declined }
        }
        rw_ba_rehome_effect = yes
        if = { limit = { NOT = { exists = scope:rw_ba_custodian } } rw_ba_retire_effect = yes }
    }
    else = { rw_ba_retire_effect = yes }
}
