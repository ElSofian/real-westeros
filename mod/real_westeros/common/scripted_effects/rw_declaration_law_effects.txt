# PRIVATE claim_cb.on_declaration prefix only. This records the callback context,
# not proof of engine ordering, selected hook consumption or an unlawful act.
rw_dl_capture_effect = {
    if = {
        limit = { rw_dl_callback_context_trigger = yes }
        scope:attacker = { save_scope_as = rw_dl_actor }
        scope:defender = { save_scope_as = rw_dl_target }
        scope:war = { save_scope_as = rw_dl_war }
        clear_saved_scope = rw_dl_issuer
        clear_saved_scope = rw_dl_authority
        scope:attacker.liege = {
            save_scope_as = rw_dl_issuer
            primary_title = { save_scope_as = rw_dl_authority }
        }
        # Direct hierarchy as observed in the native callback, not a geographic
        # border or an inferred identity for the holder of a claimed title.
        save_scope_value_as = { name = rw_dl_peer_relation value = flag:different_liege }
        if = {
            limit = {
                exists = scope:defender.liege
                scope:attacker.liege = scope:defender.liege
            }
            save_scope_value_as = { name = rw_dl_peer_relation value = flag:same_liege }
        }
        save_scope_value_as = { name = rw_dl_rule value = flag:none }
        if = {
            limit = { scope:rw_dl_issuer = { has_realm_law_flag = vassal_all_wars_banned } }
            save_scope_value_as = { name = rw_dl_rule value = flag:all }
        }
        else_if = {
            limit = { scope:rw_dl_issuer = { has_realm_law_flag = vassal_internal_wars_banned } }
            save_scope_value_as = { name = rw_dl_rule value = flag:internal }
        }
        # Keep the authority's rule and its applicability as separate facts.
        save_scope_value_as = { name = rw_dl_ban value = flag:none }
        if = {
            limit = { scope:rw_dl_rule = flag:all }
            save_scope_value_as = { name = rw_dl_ban value = flag:all }
        }
        else_if = {
            limit = {
                scope:rw_dl_rule = flag:internal
                scope:rw_dl_peer_relation = flag:same_liege
            }
            save_scope_value_as = { name = rw_dl_ban value = flag:internal }
        }
        save_scope_value_as = { name = rw_dl_gate value = flag:no_ban }
        if = {
            limit = { rw_dl_native_permission_trigger = yes }
            save_scope_value_as = { name = rw_dl_gate value = flag:permission_required }
            if = {
                limit = { scope:rw_dl_ban = flag:none }
                save_scope_value_as = { name = rw_dl_gate value = flag:inconsistent }
            }
        }
        else_if = {
            limit = { NOT = { scope:rw_dl_ban = flag:none } }
            save_scope_value_as = { name = rw_dl_gate value = flag:native_exception }
        }
        scope:rw_dl_actor = {
            if = {
                limit = {
                    exists = scope:rw_dl_issuer
                    exists = scope:rw_dl_authority
                    NOT = {
                        any_owned_story = {
                            story_type = story_rw_declaration_law
                            story_owner = scope:rw_dl_actor
                            var:rw_dl_schema = 1
                            var:rw_dl_war = scope:rw_dl_war
                        }
                    }
                }
                clear_saved_scope = rw_dl_record
                create_story = { type = story_rw_declaration_law save_scope_as = rw_dl_record }
                if = {
                    limit = {
                        exists = scope:rw_dl_record
                        scope:rw_dl_record = {
                            story_type = story_rw_declaration_law
                            story_owner = scope:rw_dl_actor
                            NOT = { has_variable = rw_dl_serial }
                        }
                    }
                    # CK3 does not create a numeric value when changing an unset variable.
                    if = {
                        limit = { NOT = { has_variable = rw_dl_serial_counter } }
                        set_variable = { name = rw_dl_serial_counter value = 0 }
                    }
                    change_variable = { name = rw_dl_serial_counter add = 1 }
                    scope:rw_dl_record = {
                        set_variable = { name = rw_dl_schema value = 1 }
                        set_variable = { name = rw_dl_serial value = scope:rw_dl_actor.var:rw_dl_serial_counter }
                        set_variable = { name = rw_dl_phase value = flag:claim_cb_on_declaration_prefix }
                        set_variable = { name = rw_dl_attacker value = scope:rw_dl_actor }
                        set_variable = { name = rw_dl_defender value = scope:rw_dl_target }
                        set_variable = { name = rw_dl_war value = scope:rw_dl_war }
                        set_variable = { name = rw_dl_issuer value = scope:rw_dl_issuer }
                        set_variable = { name = rw_dl_authority value = scope:rw_dl_authority }
                        set_variable = { name = rw_dl_seen_on value = current_date }
                        set_variable = { name = rw_dl_peer_relation value = scope:rw_dl_peer_relation }
                        set_variable = { name = rw_dl_rule value = scope:rw_dl_rule }
                        set_variable = { name = rw_dl_ban value = scope:rw_dl_ban }
                        set_variable = { name = rw_dl_gate value = scope:rw_dl_gate }
                        set_variable = { name = rw_dl_retained value = flag:yes days = 365 }
                        set_variable = { name = rw_dl_party_schema value = 1 }
                        set_variable = { name = rw_dl_acceptance_ready value = flag:yes }
                        set_variable = { name = rw_dl_claimant_state value = flag:missing }
                        if = {
                            limit = { exists = scope:claimant }
                            set_variable = { name = rw_dl_claimant value = scope:claimant }
                            set_variable = { name = rw_dl_claimant_state value = flag:known }
                        }
                        if = {
                            limit = { exists = scope:rw_dl_target.liege }
                            set_variable = { name = rw_dl_defender_liege value = scope:rw_dl_target.liege }
                        }
                    }
                    rw_dl_capture_targets_effect = yes
                }
            }
        }
    }
}
# Original CB callback context only. Capture all supplied target-title facts once.
# Counters distinguish missing/partial input from an entirely recorded title list.
rw_dl_capture_targets_effect = {
    if = {
        limit = {
            rw_dl_callback_context_trigger = yes
            exists = scope:rw_dl_actor
            exists = scope:rw_dl_record
            exists = scope:rw_dl_war
            scope:rw_dl_record = {
                story_type = story_rw_declaration_law
                story_owner = scope:rw_dl_actor
                var:rw_dl_schema = 1
                var:rw_dl_party_schema = 1
                var:rw_dl_serial > 0
                var:rw_dl_attacker = scope:rw_dl_actor
                var:rw_dl_defender = scope:defender
                var:rw_dl_war = scope:rw_dl_war
                var:rw_dl_phase = flag:claim_cb_on_declaration_prefix
                var:rw_dl_seen_on = current_date
                NOT = { has_variable = rw_dl_targets_started }
            }
        }
        scope:rw_dl_record = {
            set_variable = { name = rw_dl_targets_started value = flag:yes }
            set_variable = { name = rw_dl_target_count value = 0 }
            set_variable = { name = rw_dl_target_records value = 0 }
            set_variable = { name = rw_dl_targets_state value = flag:unavailable }
        }
        scope:rw_dl_war = {
            casus_belli = {
                every_target_title = {
                    save_scope_as = rw_dt_title
                    clear_saved_scope = rw_dt_holder
                    clear_saved_scope = rw_dt_liege
                    clear_saved_scope = rw_dt_authority
                    if = {
                        limit = { exists = holder }
                        holder = {
                            save_scope_as = rw_dt_holder
                            if = {
                                limit = { exists = liege }
                                liege = {
                                    save_scope_as = rw_dt_liege
                                    if = {
                                        limit = { exists = primary_title }
                                        primary_title = { save_scope_as = rw_dt_authority }
                                    }
                                }
                            }
                        }
                    }
                    scope:rw_dl_record = { change_variable = { name = rw_dl_target_count add = 1 } }
                    scope:rw_dl_actor = {
                        if = {
                            limit = {
                                NOT = {
                                    any_owned_story = {
                                        story_type = story_rw_declaration_target
                                        story_owner = scope:rw_dl_actor
                                        var:rw_dt_schema = 1
                                        var:rw_dt_parent = scope:rw_dl_record
                                        var:rw_dt_parent_serial = scope:rw_dl_record.var:rw_dl_serial
                                        var:rw_dt_war = scope:rw_dl_war
                                        var:rw_dt_title = scope:rw_dt_title
                                    }
                                }
                            }
                            clear_saved_scope = rw_dt_record
                            create_story = { type = story_rw_declaration_target save_scope_as = rw_dt_record }
                            if = {
                                limit = {
                                    exists = scope:rw_dt_record
                                    scope:rw_dt_record = {
                                        story_type = story_rw_declaration_target
                                        story_owner = scope:rw_dl_actor
                                        NOT = { has_variable = rw_dt_schema }
                                    }
                                }
                                scope:rw_dt_record = {
                                    set_variable = { name = rw_dt_schema value = 1 }
                                    set_variable = { name = rw_dt_parent value = scope:rw_dl_record }
                                    set_variable = { name = rw_dt_parent_serial value = scope:rw_dl_record.var:rw_dl_serial }
                                    set_variable = { name = rw_dt_war value = scope:rw_dl_war }
                                    set_variable = { name = rw_dt_title value = scope:rw_dt_title }
                                    set_variable = { name = rw_dt_seen_on value = current_date }
                                    set_variable = { name = rw_dt_holder_state value = flag:missing }
                                    if = {
                                        limit = { exists = scope:rw_dt_holder }
                                        set_variable = { name = rw_dt_holder value = scope:rw_dt_holder }
                                        set_variable = { name = rw_dt_holder_state value = flag:known }
                                    }
                                    if = {
                                        limit = { exists = scope:rw_dt_liege }
                                        set_variable = { name = rw_dt_holder_liege value = scope:rw_dt_liege }
                                    }
                                    if = {
                                        limit = { exists = scope:rw_dt_authority }
                                        set_variable = { name = rw_dt_holder_authority value = scope:rw_dt_authority }
                                    }
                                }
                                scope:rw_dl_record = { change_variable = { name = rw_dl_target_records add = 1 } }
                            }
                        }
                    }
                }
            }
        }
        scope:rw_dl_record = {
            if = {
                limit = { var:rw_dl_target_count > 0 }
                set_variable = { name = rw_dl_targets_state value = flag:partial }
                if = {
                    limit = { var:rw_dl_target_records = var:rw_dl_target_count }
                    set_variable = { name = rw_dl_targets_state value = flag:all_titles_recorded }
                }
            }
        }
    }
}
# Native on_accept tail only. Bind supplied interaction identities to an existing
# early receipt, without inventing a title-holder or claimant identity.
rw_dl_bind_acceptance_effect = {
    if = {
        limit = {
            exists = scope:actor
            exists = scope:recipient
            this = scope:actor
            is_alive = yes
            is_adult = yes
            is_landed = yes
            government_has_flag = government_is_feudal
            exists = liege
            liege = { is_alive = yes is_landed = yes }
            NOT = {
                any_character_war = {
                    count >= 2
                    primary_attacker = scope:actor
                    primary_defender = scope:recipient
                }
            }
            any_owned_story = { rw_dl_acceptance_match_trigger = yes }
            NOT = { any_owned_story = { count >= 2 rw_dl_acceptance_match_trigger = yes } }
        }
        clear_saved_scope = rw_da_record
        random_owned_story = {
            limit = { rw_dl_acceptance_match_trigger = yes }
            save_scope_as = rw_da_record
        }
        if = {
            limit = {
                exists = scope:rw_da_record
                scope:rw_da_record = {
                    rw_dl_acceptance_match_trigger = yes
                    has_variable = rw_dl_acceptance_ready
                    NOT = { has_variable = rw_dl_accepted_on }
                }
            }
            scope:rw_da_record = {
                # Consume before the remaining metadata writes.
                remove_variable = rw_dl_acceptance_ready
                set_variable = { name = rw_dl_accepted_on value = current_date }
                set_variable = { name = rw_dl_acceptance_schema value = 1 }
                set_variable = { name = rw_xw_route_ready value = flag:yes }
                set_variable = { name = rw_dl_accepted_actor value = scope:actor }
                set_variable = { name = rw_dl_accepted_recipient value = scope:recipient }
                set_variable = { name = rw_dl_accepted_choice value = flag:ordinary }
                if = {
                    limit = { always = scope:rw_uw_defy }
                    set_variable = { name = rw_dl_accepted_choice value = flag:defy_selected }
                    if = {
                        limit = { always = scope:hook }
                        set_variable = { name = rw_dl_accepted_choice value = flag:ambiguous }
                    }
                }
                else_if = {
                    limit = { always = scope:hook }
                    set_variable = { name = rw_dl_accepted_choice value = flag:hook_selected }
                }
            }
        }
    }
}
# Attach ONLY an already captured unique receipt for the exact parties/war/issuer.
# No capture fallback if the native CB callback has not run or lacks context.
rw_dl_link_response_effect = {
    if = {
        limit = {
            this = scope:rw_uw_attacker
            scope:rw_uw_record = {
                story_type = story_rw_unlawful_war_response
                story_owner = scope:rw_uw_attacker
                var:rw_uw_serial = scope:rw_uw_serial
                var:rw_uw_war = scope:rw_uw_war
                var:rw_uw_attacker = scope:rw_uw_attacker
                var:rw_uw_defender = scope:rw_uw_defender
                var:rw_uw_issuer = scope:rw_uw_issuer
                var:rw_uw_authority = scope:rw_uw_authority
                var:rw_uw_source = flag:native_choice_accept
                has_variable = rw_uw_live
                var:rw_uw_state = flag:pending
                NOT = { has_variable = rw_dl_link }
            }
            any_owned_story = { rw_dl_match_response_trigger = yes }
            NOT = { any_owned_story = { count >= 2 rw_dl_match_response_trigger = yes } }
        }
        clear_saved_scope = rw_dl_link_source
        random_owned_story = {
            limit = { rw_dl_match_response_trigger = yes }
            save_scope_as = rw_dl_link_source
        }
        if = {
            limit = {
                exists = scope:rw_dl_link_source
                scope:rw_dl_link_source = { rw_dl_match_response_trigger = yes }
            }
            scope:rw_uw_record = {
                set_variable = { name = rw_dl_link value = scope:rw_dl_link_source }
                set_variable = { name = rw_dl_link_serial value = scope:rw_dl_link_source.var:rw_dl_serial }
            }
        }
    }
}
