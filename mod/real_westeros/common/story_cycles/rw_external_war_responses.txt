# Separate personal choices, no hereditary guilt or inherited authority.
story_rw_external_war_response = {
    on_owner_death = { end_story = yes }
    effect_group = {
        days = 30
        triggered_effect = {
            trigger = { NOT = { rw_xw_exact_war_trigger = yes } }
            effect = { end_story = yes }
        }
        triggered_effect = {
            trigger = {
                rw_xw_exact_war_trigger = yes
                NOT = { var:rw_xw_state = flag:closed }
                OR = {
                    NOT = { rw_xw_context_trigger = yes }
                    AND = {
                        var:rw_xw_state = flag:pending
                        NOT = { has_variable = rw_xw_window }
                    }
                    AND = {
                        var:rw_xw_state = flag:demanded
                        NOT = { has_variable = rw_xw_answer_window }
                    }
                    AND = {
                        var:rw_xw_state = flag:native_queued
                        NOT = { has_variable = rw_xw_native_window }
                    }
                    AND = {
                        var:rw_xw_state = flag:mediation_requested
                        OR = {
                            NOT = { has_variable = rw_xm_window }
                            NOT = { rw_xm_context_trigger = yes }
                        }
                    }
                }
            }
            effect = {
                set_variable = { name = rw_xw_state value = flag:closed }
                set_variable = { name = rw_xw_result value = flag:expired_or_context_lost }
                remove_variable = rw_xw_window
                remove_variable = rw_xw_answer_window
                remove_variable = rw_xw_native_window
                remove_variable = rw_xm_window
            }
        }
    }
}
