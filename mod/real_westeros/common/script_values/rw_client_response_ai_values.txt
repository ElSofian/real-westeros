# Missing or unknown historical capacity stays neutral, never reconstructed.
rw_ai_gold_response_accountability_factor = {
    value = 1
    if = {
        limit = { rw_ai_gold_breach_context_trigger = yes }
        if = {
            limit = { scope:rw_service_provider = { has_variable = rw_service_last_performance_capacity } }
            if = {
                limit = { scope:rw_service_provider = { var:rw_service_last_performance_capacity = flag:unavailable } }
                multiply = 0.5
            }
            else_if = {
                limit = { scope:rw_service_provider = { has_variable = rw_service_last_reason var:rw_service_last_performance_capacity = flag:available } }
                if = {
                    limit = { scope:rw_service_provider = { var:rw_service_last_reason = flag:refusal } }
                    multiply = 1.25
                }
            }
        }
    }
}
rw_ai_favor_response_accountability_factor = {
    value = 1

            limit = { scope:rw_fv_provider = { has_variable = rw_fr_capacity } }
            if = { limit = { scope:rw_fv_provider = { var:rw_fr_capacity = flag:unavailable } } multiply = 0.5 }
            else_if = { limit = { scope:rw_fv_provider = { var:rw_fr_capacity = flag:available } } multiply = 1.25 }
        }
    }
}
