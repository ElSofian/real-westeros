# Own attention/capacity only. This does not make a free political choice costly.
rw_ai_xw_workload_factor = {
    value = 1
    if = { limit = { is_at_war = yes } multiply = 0.75 }
    if = { limit = { stress_level >= 2 } multiply = 0.75 }

# Current decider's own public relationships, never another character's opinions.
rw_ai_xw_attacker_favor_factor = {
    value = 1
    if = { limit = { opinion = { target = scope:rw_xw_attacker value >= 25 } } multiply = 1.25 }
    if = { limit = { opinion = { target = scope:rw_xw_attacker value <= -25 } } multiply = 0.75 }
    if = {
        limit = { OR = { is_close_family_of = scope:rw_xw_attacker is_allied_to = scope:rw_xw_attacker } }
        multiply = 1.25
    }
    min = 0.5
    max = 2
}
# Declaring liege considers the primary defender. Protector considers its actual
# representative title holder, whose exact current identity is already guarded.
# Public current stakes among this protector's exact recorded target titles.
# Each signal is presence-based. Only the highest matching rank contributes.
rw_ai_xw_protected_title_factor = {
    value = 1
    if = {
        limit = { rw_ai_xw_title_context_trigger = yes }
        if = {
            limit = {
                scope:rw_xw_attacker = {
                    any_owned_story = {
                        rw_ai_xw_protected_title_fact_trigger = yes
                        var:rw_dt_title = { tier >= tier_empire }
                    }
                }
            }
            add = 0.4
        }
        else_if = {
            limit = {
                scope:rw_xw_attacker = {
                    any_owned_story = {
                        rw_ai_xw_protected_title_fact_trigger = yes
                        var:rw_dt_title = { tier >= tier_kingdom }
                    }
                }
            }
            add = 0.3
        }
        else_if = {
            limit = {
                scope:rw_xw_attacker = {
                    any_owned_story = {
                        rw_ai_xw_protected_title_fact_trigger = yes
                        var:rw_dt_title = { tier >= tier_duchy }
                    }
                }
            }
            add = 0.2
        }
        else_if = {
            limit = {
                scope:rw_xw_attacker = {
                    any_owned_story = {
                        rw_ai_xw_protected_title_fact_trigger = yes
                        var:rw_dt_title = { tier >= tier_county }
                    }
                }
            }
            add = 0.1
        }
        if = {
            limit = {
                scope:rw_xw_attacker = {
                    any_owned_story = {
                        rw_ai_xw_protected_title_fact_trigger = yes
                        exists = var:rw_dt_holder.primary_title
                        var:rw_dt_title = { this = prev.var:rw_dt_holder.primary_title }
                    }
                }
            }
            add = 0.2
        }
        if = {
            limit = {
                scope:rw_xw_attacker = {
                    any_owned_story = {
                        count >= 2
                        rw_ai_xw_protected_title_fact_trigger = yes
                    }
                }
            }
            add = 0.1
        }
    }
    min = 1
    max = 1.7
}

# Attachment to the known original claim goal, without predicting the native result.

    }

rw_ai_xw_concern_factor = {
    value = 1
    if = {
        limit = { scope:rw_xw_record = { var:rw_xw_role = flag:declaring_liege } }
        if = { limit = { opinion = { target = scope:rw_xw_defender value >= 25 } } multiply = 1.25 }
        if = { limit = { opinion = { target = scope:rw_xw_defender value <= -25 } } multiply = 0.75 }
        if = {
            limit = { OR = { is_close_family_of = scope:rw_xw_defender is_allied_to = scope:rw_xw_defender } }
            multiply = 1.25
        }
        multiply = rw_ai_xw_external_title_factor
    }
    else = {
        if = { limit = { opinion = { target = scope:rw_xw_record.var:rw_xw_target.var:rw_dt_holder value >= 25 } } multiply = 1.25 }
        if = { limit = { opinion = { target = scope:rw_xw_record.var:rw_xw_target.var:rw_dt_holder value <= -25 } } multiply = 0.75 }
        if = {
            limit = {
                OR = {
                    is_close_family_of = scope:rw_xw_record.var:rw_xw_target.var:rw_dt_holder
                    is_allied_to = scope:rw_xw_record.var:rw_xw_target.var:rw_dt_holder
                }
            }
            multiply = 1.25
        }
        multiply = rw_ai_xw_protected_title_factor
    }
    min = 0.5
    max = 2
}

# Only the primary attacker reads its own war position for its public reply.
# Presence of living secondary participants with positive native current strength.
# Both camps use the exact same war. This is a support signal, without troop sums.
rw_ai_xw_committed_support_factor = {
    value = 1
    if = {
        limit = { rw_xw_answer_turn_trigger = yes }
        if = {
            limit = {
                scope:rw_xw_record.var:rw_xw_war = {
                    any_war_attacker = {
                        is_alive = yes
                        NOT = { this = scope:rw_xw_attacker }
                        current_military_strength > 0
                    }
                }
            }
            multiply = 1.25
        }
        if = {
            limit = {
                scope:rw_xw_record.var:rw_xw_war = {
                    any_war_defender = {
                        is_alive = yes
                        NOT = { this = scope:rw_xw_defender }
                        current_military_strength > 0
                    }
                }
            }
            divide = 1.25
        }
    }
    min = 0.8
    max = 1.25
}
rw_ai_xw_refusal_position_factor = {

    }
    min = 0.5
    max = 1.5
}
