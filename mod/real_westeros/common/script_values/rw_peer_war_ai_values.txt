# Peer-war policy. Shared xw personality/workload factors read only the decider.
# All counterpart/record references here use the peer event's original uw frame.
rw_ai_uw_attacker_favor_factor = {
    value = 1
    if = { limit = { opinion = { target = scope:rw_uw_attacker value >= 25 } } multiply = 1.25 }
    if = { limit = { opinion = { target = scope:rw_uw_attacker value <= -25 } } multiply = 0.75 }
    if = {
        limit = { OR = { is_close_family_of = scope:rw_uw_attacker is_allied_to = scope:rw_uw_attacker } }
        multiply = 1.25
    }
    min = 0.5
    max = 2
}
# Read only exact defender-side titles. Highest rank, primary title and multiple
# valid facts use the same provisional increments as the protector policy.
rw_ai_uw_title_stakes_factor = {
    value = 1
    if = {
        limit = { rw_ai_uw_title_context_trigger = yes }
        if = {
            limit = {
                scope:rw_uw_attacker = {
                    any_owned_story = {
                        rw_ai_uw_title_fact_trigger = yes
                        var:rw_dt_title = { tier >= tier_empire }
                    }
                }
            }
            add = 0.4
        }
        else_if = {
            limit = {
                scope:rw_uw_attacker = {
                    any_owned_story = {
                        rw_ai_uw_title_fact_trigger = yes
                        var:rw_dt_title = { tier >= tier_kingdom }
                    }
                }
            }
            add = 0.3
        }
        else_if = {
            limit = {
                scope:rw_uw_attacker = {
                    any_owned_story = {
                        rw_ai_uw_title_fact_trigger = yes
                        var:rw_dt_title = { tier >= tier_duchy }
                    }
                }
            }
            add = 0.2
        }
        else_if = {
            limit = {
                scope:rw_uw_attacker = {
                    any_owned_story = {
                        rw_ai_uw_title_fact_trigger = yes
                        var:rw_dt_title = { tier >= tier_county }
                    }
                }
            }
            add = 0.1
        }
        if = {
            limit = {
                scope:rw_uw_attacker = {
                    any_owned_story = {
                        rw_ai_uw_title_fact_trigger = yes
                        exists = var:rw_dt_holder.primary_title
                        var:rw_dt_title = { this = prev.var:rw_dt_holder.primary_title }
                    }
                }
            }
            add = 0.2
        }
        if = {
            limit = {
                scope:rw_uw_attacker = {
                    any_owned_story = {
                        count >= 2
                        rw_ai_uw_title_fact_trigger = yes
                    }
                }
            }
            add = 0.1
        }
    }
    min = 1
    max = 1.7
}

# Neutral for own claims, missing facts and invalid/foreign response frames.
# Only the current decider's own relationships to the recorded third party enter.
rw_ai_uw_beneficiary_favor_factor = {
    value = 1
    if = {
        limit = { rw_ai_uw_beneficiary_context_trigger = yes }
        if = { limit = { this = scope:rw_uw_record.var:rw_dl_link.var:rw_dl_claimant } multiply = 1.5 }
        else = {
            if = { limit = { opinion = { target = scope:rw_uw_record.var:rw_dl_link.var:rw_dl_claimant value >= 25 } } multiply = 1.25 }
            if = { limit = { opinion = { target = scope:rw_uw_record.var:rw_dl_link.var:rw_dl_claimant value <= -25 } } multiply = 0.75 }
            if = {
                limit = { OR = { is_close_family_of = scope:rw_uw_record.var:rw_dl_link.var:rw_dl_claimant is_allied_to = scope:rw_uw_record.var:rw_dl_link.var:rw_dl_claimant } }
                multiply = 1.25
            }
        }
    }
    min = 0.5
    max = 2
}
rw_ai_uw_resolution_factor = {
    value = 1
    multiply = rw_ai_xw_honor_factor
    multiply = rw_ai_xw_compassion_factor
    multiply = rw_ai_xw_initiative_factor
    multiply = rw_ai_uw_defender_concern_factor
    divide = rw_ai_uw_attacker_favor_factor
    divide = rw_ai_uw_beneficiary_favor_factor
}
# The original primary attacker knows the position of its own exact native war.
# Presence of living secondary participants with positive native current strength.
# Both camps use the exact same war, with a reduced signal when all supporters have other wars.
rw_ai_uw_committed_support_factor = {
    value = 1
    if = {
        limit = { rw_uw_answer_turn_trigger = yes }
        if = {
            limit = {
                scope:rw_uw_war = {
                    any_war_attacker = {
                        is_alive = yes
                        NOT = { this = scope:rw_uw_attacker }
                        current_military_strength > 0
                    }
                }
            }
            if = {
                limit = {
                    scope:rw_uw_war = {
                        any_war_attacker = {
                            is_alive = yes
                            NOT = { this = scope:rw_uw_attacker }
                            current_military_strength > 0
                            NOT = {
                                any_character_war = {
                                    NOT = { this = scope:rw_uw_war }
                                }
                            }
                        }
                    }
                }
                multiply = 1.25
            }
            else = { multiply = 1.1 }
        }
        if = {
            limit = {
                scope:rw_uw_war = {
                    any_war_defender = {
                        is_alive = yes
                        NOT = { this = scope:rw_uw_defender }
                        current_military_strength > 0
                    }
                }
            }
            if = {
                limit = {
                    scope:rw_uw_war = {
                        any_war_defender = {
                            is_alive = yes
                            NOT = { this = scope:rw_uw_defender }
                            current_military_strength > 0
                            NOT = {
                                any_character_war = {
                                    NOT = { this = scope:rw_uw_war }
                                }
                            }
                        }
                    }
                }
                divide = 1.25
            }
            else = { divide = 1.1 }
        }
    }
    min = 0.8
    max = 1.25
}
# Own army co-location supplies the observation anchor for grouped reinforcements.
rw_ai_uw_observed_reinforcement_factor = {
    value = 1
    if = {
        limit = { rw_uw_answer_turn_trigger = yes }
        if = {
            limit = {
                scope:rw_uw_attacker = {
                    NOT = {
                        any_character_war = { NOT = { this = scope:rw_uw_war } }
                    }
                }
                scope:rw_uw_war = {
                    any_war_attacker = {
                        is_alive = yes
                        NOT = { this = scope:rw_uw_attacker }
                        any_army = {
                            army_owner = prev
                            army_size > 0
                            NOT = { is_raid_army = yes }
                            exists = location
                            location = {
                                any_army_in_location = {
                                    army_owner = scope:rw_uw_attacker
                                    army_size > 0
                                    NOT = { is_raid_army = yes }
                                }
                            }
                        }
                    }
                }
            }
            multiply = 1.1
        }
    }
    min = 1
    max = 1.1
}

        divide = rw_ai_xw_rationality_factor
    }
    min = 0.5
    max = 1.5
}
