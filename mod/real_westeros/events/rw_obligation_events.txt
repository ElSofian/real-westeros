rw_obligations.0001 = {
    type = character_event
    title = rw_obligations.0001.t
    desc = {
        desc = rw_obligations.0001.desc
        triggered_desc = { trigger = { scope:rw_service_price = 25 } desc = rw_delivery.price_25 }
        triggered_desc = { trigger = { scope:rw_service_price = 50 } desc = rw_delivery.price_50 }
        triggered_desc = { trigger = { rw_service_short_delivery_trigger = yes } desc = rw_delivery.days_14 }
        triggered_desc = { trigger = { rw_service_long_delivery_trigger = yes } desc = rw_delivery.days_30 }
    }
    theme = diplomacy
    left_portrait = { character = scope:rw_service_client }
    option = {
        name = rw_obligations.fulfill
        trigger = { rw_service_obligation_fulfillable_trigger = yes }
        rw_service_fulfill_promise_effect = yes
        ai_chance = { base = 70 }
    }
    option = {
        name = rw_obligations.break
        trigger = { rw_service_obligation_actionable_trigger = yes }
        rw_service_break_promise_effect = yes
        ai_chance = { base = 10 }
    }
    option = {
        name = rw_cancellations.propose
        trigger = {
            rw_cancel_event_can_propose_trigger = { PARTY = scope:rw_service_provider }
            rw_service_obligation_actionable_trigger = yes
        }
        rw_cancel_provider_propose_effect = yes
        # A pending/rejected request never stops the ordinary due-date revisit.
        rw_service_wait_promise_effect = yes
        ai_chance = {
            base = 5
            modifier = { factor = 3 is_at_war = yes }
        }
    }
    option = {
        name = rw_amendments.propose
        trigger = { rw_service_obligation_actionable_trigger = yes rw_amend_can_start_trigger = yes this = scope:rw_service_provider }
        # Replace the current ticket before sending; give a short-contract provider another chance.
        rw_service_schedule_decision_effect = { DAYS = 3 }
        rw_amend_provider_propose_effect = yes
        ai_chance = { base = 10 modifier = { factor = 2 has_trait = generous } }
    }
    option = {
        name = rw_obligations.wait
        trigger = { rw_service_obligation_actionable_trigger = yes }
        # Leave it live and revisit in a week, for AI and human alike.
        rw_service_wait_promise_effect = yes
        ai_chance = { base = 20 }
    }
    option = {
        name = rw_obligations.lapsed
        trigger = { NOT = { rw_service_obligation_actionable_trigger = yes } }
        # Never clear a live obligation just because this popup is obsolete/early.
        ai_chance = { base = 100 }
    }
}

# Only the current reminder ticket may create a visible provider decision.
rw_obligations.0004 = {
    type = character_event
    hidden = yes
    immediate = {
        if = {
            limit = {
                rw_service_obligation_due_trigger = yes
                rw_service_decision_matches_trigger = yes
                has_character_flag = rw_service_reminder_pending
            }
            remove_character_flag = rw_service_reminder_pending
            add_character_flag = rw_service_decision_open
            trigger_event = { id = rw_obligations.0001 }
        }
    }
}
