namespace = rw_scheme_response

# Observer opportunity, automatically offered once or explicitly opened from the journal.
rw_scheme_response.0001 = {
    type = character_event
    title = rw_scheme_response.owner_title
    desc = {
        first_valid = {
            triggered_desc = { trigger = { rw_scheme_response_owner_turn_trigger = yes } desc = rw_scheme_response.owner_desc }
            triggered_desc = { trigger = { rw_scheme_response_can_withdraw_trigger = yes } desc = rw_scheme_response.withdraw_desc }
            desc = rw_scheme_response.unavailable
        }
    }
    theme = intrigue
    option = {
        name = rw_scheme_response.forego
        trigger = { rw_scheme_response_owner_turn_trigger = yes }
        rw_scheme_response_forego_effect = yes
        ai_chance = {
            base = 40
            modifier = { factor = 2 is_at_war = yes }
            modifier = { factor = 2 opinion = { target = scope:rw_scheme_participant value > 25 } }
        }
    }
    option = {
        name = rw_scheme_response.request
        trigger = { rw_scheme_response_can_request_trigger = yes }
        rw_scheme_response_request_effect = yes
        ai_chance = {
            base = 30
            modifier = { factor = 2 opinion = { target = scope:rw_scheme_participant value < -25 } }
        }
    }
    option = {
        name = rw_scheme_response.withdraw
        trigger = { rw_scheme_response_can_withdraw_trigger = yes }
        rw_scheme_response_withdraw_effect = yes
        ai_chance = {
            base = 30
            modifier = { factor = 2 is_at_war = yes }
            modifier = { factor = 2 opinion = { target = scope:rw_scheme_participant value > 25 } }
        }
    }
    option = {
        name = rw_scheme_response.later
        trigger = { OR = { rw_scheme_response_owner_turn_trigger = yes rw_scheme_response_can_withdraw_trigger = yes } }
        # Leave state and deadline alone; no new reminder or automatic response.
        ai_chance = { base = 30 }
    }
    option = {
        name = rw_scheme_response.acknowledge
        trigger = { NOT = { OR = { rw_scheme_response_owner_turn_trigger = yes rw_scheme_response_can_withdraw_trigger = yes } } }
        ai_chance = { base = 100 }
    }
}
