namespace = rw_services

# Client receives a 50-gold quotation; accepting performs the endorsement immediately.
rw_services.0001 = {
    type = character_event
    title = rw_services.0001.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { rw_service_deferred_trigger = yes }
                desc = {
                    desc = rw_services.0001.deferred_desc
                    triggered_desc = { trigger = { rw_service_short_delivery_trigger = yes } desc = rw_delivery.days_14 }
                    triggered_desc = { trigger = { rw_service_long_delivery_trigger = yes } desc = rw_delivery.days_30 }
                }
            }
            desc = rw_services.0001.desc
        }
    }
    theme = diplomacy
    left_portrait = { character = scope:rw_service_provider }
    option = {
        name = rw_services.accept_50
        trigger = { rw_service_client_turn_trigger = yes rw_service_affordable_trigger = yes NOT = { rw_service_deferred_trigger = yes } }
        rw_service_settle_effect = yes
        ai_chance = { base = 0 modifier = { add = rw_ai_service_accept_weight always = yes } }
    }
    option = {
        name = rw_services.promise_50
        trigger = { rw_service_client_turn_trigger = yes rw_service_deferred_trigger = yes rw_service_affordable_trigger = yes }
        rw_service_accept_promise_effect = yes
        ai_chance = { base = 0 modifier = { add = rw_ai_service_accept_weight always = yes } }
    }
    option = {
        name = rw_services.counter_25
        trigger = { rw_service_client_turn_trigger = yes scope:rw_service_client = { gold >= 25 } }
        rw_service_counter_effect = yes
        ai_chance = { base = 40 }
    }
    option = {
        name = rw_delivery.shorten
        trigger = { rw_service_can_shorten_delivery_trigger = yes }
        rw_service_shorten_delivery_effect = yes
        ai_chance = { base = 20 }
    }
    option = {
        name = rw_services.refuse
        trigger = { rw_service_client_turn_trigger = yes }
        rw_service_clear_effect = yes
        ai_chance = { base = 30 }
    }
    option = {
        name = rw_services.lapsed
        trigger = { NOT = { rw_service_client_turn_trigger = yes } }
        rw_service_clear_effect = yes
        ai_chance = { base = 100 }
    }
}
